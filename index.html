function toggleHabitLog(habit, dateStr, isChecked) { // If checked, status is 'Done'. If unchecked, status is 'Missed' (or blank/removed) const status = isChecked ? 'Done' : 'Missed'; updateSyncBadge('● Saving...'); const existingIndex = logs.findIndex(l => l.date === dateStr && l.habit === habit); if (existingIndex >= 0) { logs[existingIndex].status = status; } else { logs.push({ date: dateStr, habit: habit, status: status }); } localStorage.setItem('hb_logs', JSON.stringify(logs)); renderHabitTable(); updateKPIs(); renderCharts(); fetch(SCRIPT_URL, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'saveLog', date: dateStr, habit: habit, status: status }) }) .then(() => updateSyncBadge('● Synced')) .catch(() => updateSyncBadge('● Offline', true)); }
